CREATE TABLE Brands (
    brand_id            NUMBER          PRIMARY KEY,
    brand_name          VARCHAR2(100)   NOT NULL,
    brand_logo_url      VARCHAR2(255),
    brand_description   VARCHAR2(1000),
    brand_origin        VARCHAR2(100)
);

CREATE TABLE SubBrand (
    sub_brand_id      NUMBER PRIMARY KEY,
    brand_id          NUMBER NOT NULL,
    sub_brand_name    VARCHAR2(100) NOT NULL,
    sub_brand_logo_url VARCHAR2(255),

    CONSTRAINT fk_subbrand_brand
        FOREIGN KEY (brand_id)
        REFERENCES Brands(brand_id)
);

CREATE TABLE Product (
    product_id               NUMBER          PRIMARY KEY,
    brand_id                 NUMBER          NOT NULL,
    sub_brand_id             NUMBER          NULL,

    product_name             VARCHAR2(150)   NOT NULL,
    product_volume           VARCHAR2(50),
    product_image_url        VARCHAR2(255),
    product_nutritional_info CLOB,
    product_launch_date      DATE,
    product_description      VARCHAR2(1000),

    CONSTRAINT fk_product_brand
        FOREIGN KEY (brand_id)
        REFERENCES Brands(brand_id),

    CONSTRAINT fk_product_sub_brand
        FOREIGN KEY (sub_brand_id)
        REFERENCES SubBrand(sub_brand_id)
);

CREATE TABLE Product_Video (
    video_id        NUMBER PRIMARY KEY,
    product_id      NUMBER NOT NULL,

    youtube_video_id VARCHAR2(50) NOT NULL, -- TiKjZ6vQZIU
    video_title      VARCHAR2(200),
    video_description VARCHAR2(1000),

    CONSTRAINT fk_video_product
        FOREIGN KEY (product_id)
        REFERENCES Product(product_id)
        ON DELETE CASCADE
);


CREATE TABLE Category (
    brand_category_id          NUMBER          PRIMARY KEY,
    brand_category_name        VARCHAR2(100)   NOT NULL,
    brand_category_description VARCHAR2(500)
);

CREATE TABLE Brand_Category (
    brand_category_id  NUMBER,
    brand_id           NUMBER,
    PRIMARY KEY (brand_category_id, brand_id),
    CONSTRAINT fk_bc_category FOREIGN KEY (brand_category_id)
        REFERENCES Category(brand_category_id),
    CONSTRAINT fk_bc_brand FOREIGN KEY (brand_id)
        REFERENCES Brands(brand_id)
);
