CREATE TABLE CAMPAIGNS (
    campaigns_id        NUMBER        PRIMARY KEY,
    campaigns_title     VARCHAR2(255),
    campaigns_summary   VARCHAR2(255),
    campaigns_body      VARCHAR2(255),
    campaigns_date      DATE
);

CREATE TABLE STORIES (
    story_id        NUMBER        PRIMARY KEY,
    story_title     VARCHAR2(255),
    story_summary   VARCHAR2(255),
    story_body      CLOB,
    story_image     VARCHAR2(255),
    story_date      DATE
);

CREATE TABLE DISC_MEDIA (
    media_id        NUMBER,
    story_id        NUMBER,
    campaigns_id    NUMBER,
    media_url       VARCHAR2(255),
    media_index     NUMBER,

    -- 복합 PRIMARY KEY
    CONSTRAINT pk_disc_media 
        PRIMARY KEY (media_id, story_id, campaigns_id),

    -- FK: STORIES
    CONSTRAINT fk_discmedia_story
        FOREIGN KEY (story_id)
        REFERENCES STORIES (story_id),

    -- FK: CAMPAIGNS
    CONSTRAINT fk_discmedia_campaign
        FOREIGN KEY (campaigns_id)
        REFERENCES CAMPAIGNS (campaigns_id)
);
