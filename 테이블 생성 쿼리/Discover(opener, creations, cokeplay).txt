opener

CREATE TABLE OPENER_CATEGORY (
CATEGORY_ID      NUMBER         PRIMARY KEY,
CATEGORY_NAME    VARCHAR2(100)  NOT NULL,
SECTION_TITLE    VARCHAR2(200),
DISPLAY_ORDER    NUMBER,
USE_YN           CHAR(1)        DEFAULT 'Y'
);


CREATE TABLE OPENER_POST (
    POST_ID         NUMBER          PRIMARY KEY,
    CATEGORY_ID     NUMBER          NOT NULL,
    TITLE           VARCHAR2(300)   NOT NULL,
    SUMMARY         VARCHAR2(1000),
    THUMBNAIL_URL   VARCHAR2(500),
    EXTERNAL_URL    VARCHAR2(500),
    DETAIL_URL      VARCHAR2(500),   
    CONTENT         CLOB,                
    CREATED_AT      DATE            DEFAULT SYSDATE,
    VIEW_COUNT      NUMBER          DEFAULT 0,

    CONSTRAINT FK_OPENER_POST_CATEGORY
        FOREIGN KEY (CATEGORY_ID)
        REFERENCES OPENER_CATEGORY (CATEGORY_ID)
);


CREATE TABLE OPENER_POST_DETAIL (
DETAIL_ID     NUMBER PRIMARY KEY,
POST_ID       NUMBER NOT NULL,         -- OPENER_POST 참조

BLOCK_TYPE    VARCHAR2(20) NOT NULL,   -- 'TEXT' | 'IMAGE'
TITLE         VARCHAR2(300),            -- TEXT일 때 제목
CONTENT       CLOB,                     -- TEXT일 때 본문
IMAGE_URL     VARCHAR2(500),            -- IMAGE일 때 이미지 경로

DISPLAY_ORDER NUMBER NOT NULL,          -- 출력 순서
CREATED_AT    DATE DEFAULT SYSDATE,

CONSTRAINT FK_OPENER_POST_DETAIL
FOREIGN KEY (POST_ID)
REFERENCES OPENER_POST (POST_ID)
);


creations

CREATE TABLE CREATION_STORY (
    STORY_ID        NUMBER          PRIMARY KEY,
    SECTION_TYPE    VARCHAR2(20)    NOT NULL,   -- MAIN / RECOMMEND / MORE
    TITLE           VARCHAR2(300)   NOT NULL,
    SUMMARY         VARCHAR2(1000),
    THUMBNAIL_URL   VARCHAR2(500),
    EXTERNAL_URL    VARCHAR2(500),
    CONTENT         CLOB,              
    CREATED_AT      DATE            DEFAULT SYSDATE,
    DISPLAY_ORDER   NUMBER
);

CREATE TABLE CREATION_DETAIL (
    DETAIL_ID   NUMBER PRIMARY KEY,
    STORY_ID    NUMBER NOT NULL,
    CONTENT     CLOB NOT NULL,      -- 상세 본문
    CREATED_AT  DATE DEFAULT SYSDATE,
    
    CONSTRAINT FK_CREATION_DETAIL
    FOREIGN KEY (STORY_ID)
    REFERENCES CREATION_STORY(STORY_ID)
);
