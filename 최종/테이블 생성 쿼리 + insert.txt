-- brands 테이블 생성 쿼리 (다원)
CREATE TABLE Brands (
    brand_id            NUMBER          PRIMARY KEY,
    brand_name          VARCHAR2(100)   NOT NULL,
    brand_logo_url      VARCHAR2(255),
    brand_description   VARCHAR2(1000),
    brand_origin        VARCHAR2(100),
    brand_mainphoto_url     VARCHAR2(255)
);

CREATE TABLE Brand_Video (
    brand_id            NUMBER PRIMARY KEY,
    youtube_video_id    VARCHAR2(50) NOT NULL,
    video_thumbnail_url VARCHAR2(255) NOT NULL,
    video_description   VARCHAR2(1000),

    CONSTRAINT fk_brand_video
        FOREIGN KEY (brand_id)
        REFERENCES Brands(brand_id)
);

CREATE TABLE SubBrand (
    sub_brand_id      NUMBER PRIMARY KEY,
    brand_id          NUMBER NOT NULL,
    sub_brand_name    VARCHAR2(100) NOT NULL,
    sub_brand_logo_url VARCHAR2(255),

    CONSTRAINT fk_subbrand_brand
        FOREIGN KEY (brand_id)
        REFERENCES Brands(brand_id)
);

CREATE TABLE Product (
    product_id        NUMBER          PRIMARY KEY,
    brand_id          NUMBER          NOT NULL,
    sub_brand_id      NUMBER,

    product_name      VARCHAR2(150)   NOT NULL,
    product_volume    VARCHAR2(50),
    product_image_url VARCHAR2(255),
    product_description VARCHAR2(1000),

    CONSTRAINT fk_product_brand
        FOREIGN KEY (brand_id)
        REFERENCES Brands(brand_id),

    CONSTRAINT fk_product_sub_brand
        FOREIGN KEY (sub_brand_id)
        REFERENCES SubBrand(sub_brand_id)
);

CREATE TABLE Category (
    brand_category_id          NUMBER          PRIMARY KEY,
    brand_category_name        VARCHAR2(100)   NOT NULL,
    brand_category_description VARCHAR2(500)
);

CREATE TABLE Brand_Category (
    brand_category_id  NUMBER,
    brand_id           NUMBER,
    PRIMARY KEY (brand_category_id, brand_id),
    CONSTRAINT fk_bc_category FOREIGN KEY (brand_category_id)
        REFERENCES Category(brand_category_id),
    CONSTRAINT fk_bc_brand FOREIGN KEY (brand_id)
        REFERENCES Brands(brand_id)
);

-- ourcompany 테이블 생성 쿼리 (나영)
CREATE TABLE OURCOMPANY (
    CONTENT_ID      NUMBER          PRIMARY KEY,       
    SECTION_CODE    VARCHAR2(50)    NOT NULL,        
    TITLE           VARCHAR2(255)   NOT NULL,     
    SUMMARY         VARCHAR2(1000)  NOT NULL,        
    IMAGE_FILENAME  VARCHAR2(255)   NOT NULL,        
    BUTTON_LINK     VARCHAR2(255)   DEFAULT '#',      
    DISPLAY_ORDER   NUMBER          NOT NULL          
);

CREATE SEQUENCE SEQ_OURCOMPANY START WITH 1 INCREMENT BY 1;

    coke_meal_summary   VARCHAR2(255),

CREATE TABLE ORIGINALS (

CREATE TABLE BRAND_STORY (

--opener

CREATE TABLE OPENER_CATEGORY (
CATEGORY_ID      NUMBER         PRIMARY KEY,
CATEGORY_NAME    VARCHAR2(100)  NOT NULL,
SECTION_TITLE    VARCHAR2(200),
DISPLAY_ORDER    NUMBER,
USE_YN           CHAR(1)        DEFAULT 'Y'
);


CREATE TABLE OPENER_POST (
    POST_ID         NUMBER          PRIMARY KEY,
    CATEGORY_ID     NUMBER          NOT NULL,
    TITLE           VARCHAR2(300)   NOT NULL,
    SUMMARY         VARCHAR2(1000),
    THUMBNAIL_URL   VARCHAR2(500),
    EXTERNAL_URL    VARCHAR2(500),
    DETAIL_URL      VARCHAR2(500),   
    CONTENT         CLOB,                
    CREATED_AT      DATE            DEFAULT SYSDATE,
    VIEW_COUNT      NUMBER          DEFAULT 0,

    CONSTRAINT FK_OPENER_POST_CATEGORY
        FOREIGN KEY (CATEGORY_ID)
        REFERENCES OPENER_CATEGORY (CATEGORY_ID)
);


CREATE TABLE OPENER_POST_DETAIL (
DETAIL_ID     NUMBER PRIMARY KEY,
POST_ID       NUMBER NOT NULL,         -- OPENER_POST 참조

BLOCK_TYPE    VARCHAR2(20) NOT NULL,   -- 'TEXT' | 'IMAGE'
TITLE         VARCHAR2(300),            -- TEXT일 때 제목
CONTENT       CLOB,                     -- TEXT일 때 본문
IMAGE_URL     VARCHAR2(500),            -- IMAGE일 때 이미지 경로

DISPLAY_ORDER NUMBER NOT NULL,          -- 출력 순서
CREATED_AT    DATE DEFAULT SYSDATE,

CONSTRAINT FK_OPENER_POST_DETAIL
FOREIGN KEY (POST_ID)
REFERENCES OPENER_POST (POST_ID)
);


--creations

CREATE TABLE CREATION_STORY (
    STORY_ID        NUMBER          PRIMARY KEY,
    SECTION_TYPE    VARCHAR2(20)    NOT NULL,   -- MAIN / RECOMMEND / MORE
    TITLE           VARCHAR2(300)   NOT NULL,
    SUMMARY         VARCHAR2(1000),
    THUMBNAIL_URL   VARCHAR2(500),
    EXTERNAL_URL    VARCHAR2(500),
    CONTENT         CLOB,              
    CREATED_AT      DATE            DEFAULT SYSDATE,
    DISPLAY_ORDER   NUMBER
);

CREATE TABLE CREATION_DETAIL (
    DETAIL_ID   NUMBER PRIMARY KEY,
    STORY_ID    NUMBER NOT NULL,
    CONTENT     CLOB NOT NULL,      -- 상세 본문
    CREATED_AT  DATE DEFAULT SYSDATE,
    
    CONSTRAINT FK_CREATION_DETAIL
    FOREIGN KEY (STORY_ID)
    REFERENCES CREATION_STORY(STORY_ID)
);

CREATE TABLE COKEPLAY_BENEFIT (

    TITLE           VARCHAR2(300)   NOT NULL,
    DESCRIPTION     VARCHAR2(1000),
    IMAGE_URL       VARCHAR2(500),
    EXTERNAL_URL    VARCHAR2(500),
    DISPLAY_ORDER   NUMBER,

);

-- sustainability 테이블 생성 쿼리 (승찬)
CREATE TABLE SECTION (
    section_id        NUMBER          PRIMARY KEY,
    section_title     VARCHAR2(100),
    section_subtitle  VARCHAR2(100),
    section_description CLOB,
    section_image     VARCHAR2(255)
);

CREATE TABLE CONTENT (
    content_id        NUMBER          PRIMARY KEY,
    section_id        NUMBER,
    content_title     VARCHAR2(100),
    content_body      CLOB,
    content_category  VARCHAR2(50),
    content_link      VARCHAR2(200),
    DETAIL_CONTENT_ID   NUMBER , 
    CONSTRAINT fk_content_section
        FOREIGN KEY (section_id)
        REFERENCES SECTION(section_id)
);

CREATE TABLE MEDIA (
    media_id          NUMBER          PRIMARY KEY,
    content_id        NUMBER,
    media_file_type   VARCHAR2(50),
    media_file        VARCHAR2(100),
    media_description CLOB,
    CONSTRAINT fk_media_content
        FOREIGN KEY (content_id)
        REFERENCES CONTENT(content_id)
);

CREATE TABLE DOCUMENT (
    document_id         NUMBER        PRIMARY KEY,
    section_id          NUMBER,
    document_title      VARCHAR2(100),
    document_file       VARCHAR2(100),
    document_description CLOB,
    document_date       DATE,
    CONSTRAINT fk_document_section
        FOREIGN KEY (section_id)
        REFERENCES SECTION(section_id)

);
